# =====================================================================
# Caddy auth proxy for Ollama
# =====================================================================
# This sits in front of Ollama and requires a bearer token on every
# request. After validating, it strips the Authorization header before
# forwarding upstream — Ollama doesn't natively understand inbound
# bearer tokens and may 403 if it sees one it didn't issue.
#
# Replace YOUR_BEARER_KEY with the actual key generated via:
#   [Convert]::ToBase64String((1..32 | %{[byte](Get-Random -Max 256)}))
# =====================================================================

:11434 {
    # Match requests with the correct Authorization header
    @authed header Authorization "Bearer YOUR_BEARER_KEY"

    handle @authed {
        reverse_proxy host.docker.internal:11500 {
            # Strip the Authorization header before forwarding to Ollama.
            # Caddy has already validated it; Ollama would just be confused.
            header_up -Authorization

            # Long timeouts so streaming chat completions don't get cut off
            # mid-response (model generation can take 30+ seconds for big prompts).
            transport http {
                read_timeout 600s
                write_timeout 600s
            }

            # CRITICAL for streaming responses — without this, Caddy buffers
            # the entire response and you get a wall of text at the end
            # instead of token-by-token streaming.
            flush_interval -1
        }
    }

    # Anything that doesn't match @authed (no header, wrong key) → 401
    handle {
        respond "Unauthorized" 401
    }

    # Access logs to file with rotation, so the log doesn't grow forever
    log {
        output file /var/log/caddy/access.log {
            roll_size 10mb
            roll_keep 5
            roll_keep_for 720h
        }
        format json
    }
}
